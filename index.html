How to tell boards apart

Use the factory MAC address burned into eFuse. Every ESP32 has a unique one, it survives reflashing and erasing, and esptool-js reads it during the handshake before any firmware runs. The same value is available in firmware via ESP.getEfuseMac(), so the board can announce itself on Wi-Fi with the exact ID the setup page recorded.

The ID format this page uses is wrover- plus the last three MAC bytes, like wrover-a1b2c3. Friendly names ("kitchen dial") live in the registry as a label on top of that key.

Your board

The Freenove ESP32-WROVER pairs an ESP32-D0WD-V3 with 4 MB flash, 8 MB PSRAM and an OV2640 camera connector, behind a CH340 USB-UART chip. Because it is a classic ESP32, it has no native USB. Every browser connection runs through the CH340 serial bridge, which is exactly the Web Serial path in your notes. Some Windows machines need the CH340 driver before the port appears.

The "web compiler" and browser limits

  • Compiling in the browser isn't practical. The Xtensa GCC toolchain and Arduino core are hundreds of MB. Vercel functions also time out well before an ESP32 build finishes. Run arduino-cli or PlatformIO on a separate build server (a small VPS or a GitHub Action) that returns .bin files, and let the Vercel page flash them.
  • Better: compile once, configure at runtime. Ship one firmware with all capabilities, then send pins, sensor types and Wi-Fi over serial into NVS. Most "custom builds" become a JSON config, and flashing stays instant.
  • Web Serial rules. Chrome, Edge or Opera on desktop only. HTTPS required (Vercel is fine). The port picker only opens from a click. No Firefox, Safari or iOS.
  • Embedded frames. Web Serial is off inside iframes unless the parent grants allow="serial". This preview frame may not grant it, so use Simulate board here and real hardware once deployed.
  • Mixed content. The HTTPS page can't reach ws:// or http:// on your LAN. That covers your hub and the camera's MJPEG stream on port 81. Options: a cloud broker with wss:// (HiveMQ Cloud, EMQX), a tunnel (Cloudflare Tunnel) to the hub, or serve the live dashboard from the hub itself over plain HTTP.
  • Credentials over serial. Wi-Fi passwords travel in plain text over USB. That's acceptable on a local cable, but never log them. This page masks the password in the event log.

// Arduino (ESP32 core). Listens for {"cmd":"wifi",...} and announces itself.
#include <WiFi.h>
#include <Preferences.h>
#include <ArduinoJson.h>
Preferences prefs;

String deviceId() {
  uint64_t mac = ESP.getEfuseMac();           // same MAC the web page read
  char id[16];
  snprintf(id, sizeof(id), "wrover-%02x%02x%02x",
           (uint8_t)(mac >> 24), (uint8_t)(mac >> 32), (uint8_t)(mac >> 40));
  return String(id);
}

void hello() {                                // the web page parses this line
  Serial.printf("{\"id\":\"%s\",\"fw\":\"0.1.0\",\"caps\":[\"potentiometer\",\"button\",\"camera\"]}\n",
                deviceId().c_str());
}

void setup() {
  Serial.begin(115200);
  prefs.begin("net", false);
  hello();
  String ssid = prefs.getString("ssid", "");
  if (ssid.length()) WiFi.begin(ssid.c_str(), prefs.getString("pass", "").c_str());
}

void loop() {
  if (Serial.available()) {
    String line = Serial.readStringUntil('\n');
    JsonDocument doc;
    if (deserializeJson(doc, line)) return;
    if (doc["cmd"] == "wifi") {
      prefs.putString("ssid", doc["ssid"].as<String>());
      prefs.putString("pass", doc["pass"].as<String>());
      prefs.putString("hub",  doc["hub"].as<String>());
      Serial.printf("{\"ok\":true,\"id\":\"%s\",\"saved\":\"wifi\"}\n", deviceId().c_str());
      delay(200); ESP.restart();
    } else if (doc["cmd"] == "hello") hello();
  }
}

    Loading different code

    Flashing a new image replaces the running program completely, so nothing extra is needed to override this demo. Flash any other .bin in step 4 and the board reboots into it. This tab locks again because the new firmware won't identify as mpu6050-ball.

    Settings saved in NVS, like Wi-Fi, survive a normal flash. To wipe everything, use Erase entire flash in step 4.

    mpu6050_ball.ino · board "ESP32 Wrover Module" · no extra libraries
    // mpu6050_ball.ino
// Freenove ESP32-WROVER + MPU-6050 demo for WROVER Flash Lab (IMU 3D tab).
//
// Wiring (GY-521 breakout):
//   VCC -> 3V3   (the GY-521 has its own regulator, so 5V also works)
//   GND -> GND
//   SCL -> GPIO 22
//   SDA -> GPIO 21
//
// Arduino IDE: Board "ESP32 Wrover Module". No extra libraries needed (Wire is built in).
// Build:       Sketch -> Export Compiled Binary, then flash the *.merged.bin at 0x0.
//
// Serial protocol, 115200 baud, one JSON object per line:
//   board -> page  {"id":"wrover-a1b2c3","fw":"mpu6050-ball","ver":"1.0.0","caps":["imu"]}
//                  {"t":"selftest", ...hardware report...}
//                  {"t":"imu","s":123,"ax":0.01,"ay":-0.02,"az":1.00,"gx":0.3,"gy":-0.1,"gz":0.0,"tc":27.4}
//                  {"t":"err","msg":"..."}
//   page -> board  {"cmd":"hello"}  {"cmd":"selftest"}  {"cmd":"stream","on":false}

#include <Wire.h>
#include <esp_chip_info.h>

#define FW_NAME "mpu6050-ball"
#define FW_VER  "1.0.0"
#define SDA_PIN 21
#define SCL_PIN 22
#define RATE_HZ 50

// MPU-6050 registers
#define REG_SMPLRT_DIV   0x19
#define REG_CONFIG       0x1A
#define REG_GYRO_CONFIG  0x1B
#define REG_ACCEL_CONFIG 0x1C
#define REG_ACCEL_XOUT_H 0x3B
#define REG_PWR_MGMT_1   0x6B
#define REG_WHO_AM_I     0x75

const float ACCEL_LSB_PER_G  = 16384.0f;  // +-2 g range
const float GYRO_LSB_PER_DPS = 131.0f;    // +-250 deg/s range

uint8_t  mpuAddr   = 0;     // 0x68 (AD0 low) or 0x69 (AD0 high); 0 = not found
uint8_t  whoAmI    = 0;
bool     streaming = true;
uint32_t seq       = 0;
uint32_t lastSample = 0, lastErr = 0;
String   rxLine;

struct Imu { float ax, ay, az, gx, gy, gz, tc; };

String deviceId() {
  uint64_t mac = ESP.getEfuseMac();         // same factory MAC the web page reads over USB
  char id[16];
  snprintf(id, sizeof(id), "wrover-%02x%02x%02x",
           (uint8_t)(mac >> 24), (uint8_t)(mac >> 32), (uint8_t)(mac >> 40));
  return String(id);
}

bool probe(uint8_t addr) {
  Wire.beginTransmission(addr);
  return Wire.endTransmission() == 0;
}

bool writeReg(uint8_t reg, uint8_t val) {
  Wire.beginTransmission(mpuAddr);
  Wire.write(reg);
  Wire.write(val);
  return Wire.endTransmission() == 0;
}

bool readRegs(uint8_t reg, uint8_t *buf, uint8_t len) {
  Wire.beginTransmission(mpuAddr);
  Wire.write(reg);
  if (Wire.endTransmission(false) != 0) return false;
  if (Wire.requestFrom(mpuAddr, len) != len) return false;
  for (uint8_t i = 0; i < len; i++) buf[i] = Wire.read();
  return true;
}

bool initMpu() {
  mpuAddr = probe(0x68) ? 0x68 : probe(0x69) ? 0x69 : 0;
  if (!mpuAddr) return false;
  whoAmI = 0;
  readRegs(REG_WHO_AM_I, &whoAmI, 1);
  return writeReg(REG_PWR_MGMT_1, 0x01)       // wake up, clock from gyro X PLL
      && writeReg(REG_SMPLRT_DIV, 19)         // 1 kHz / (1 + 19) = 50 Hz
      && writeReg(REG_CONFIG, 0x03)           // ~44 Hz low-pass filter
      && writeReg(REG_GYRO_CONFIG, 0x00)      // +-250 deg/s
      && writeReg(REG_ACCEL_CONFIG, 0x00);    // +-2 g
}

bool readImu(Imu &m) {
  uint8_t b[14];
  if (!mpuAddr || !readRegs(REG_ACCEL_XOUT_H, b, 14)) return false;
  auto s16 = [&](int i) { return (int16_t)((b[i] << 8) | b[i + 1]); };
  m.ax = s16(0)  / ACCEL_LSB_PER_G;
  m.ay = s16(2)  / ACCEL_LSB_PER_G;
  m.az = s16(4)  / ACCEL_LSB_PER_G;
  m.tc = s16(6)  / 340.0f + 36.53f;
  m.gx = s16(8)  / GYRO_LSB_PER_DPS;
  m.gy = s16(10) / GYRO_LSB_PER_DPS;
  m.gz = s16(12) / GYRO_LSB_PER_DPS;
  return true;
}

void hello() {
  Serial.printf("{\"id\":\"%s\",\"fw\":\"" FW_NAME "\",\"ver\":\"" FW_VER "\",\"caps\":[\"imu\"]}\n",
                deviceId().c_str());
}

// Hardware report the page uses for its verification checklist.
void selftest() {
  String found;                                  // every address that ACKs on the bus
  for (uint8_t a = 1; a < 127; a++) {
    if (!probe(a)) continue;
    char h[10];
    snprintf(h, sizeof(h), "\"0x%02x\"", a);
    if (found.length()) found += ",";
    found += h;
  }

  bool mpuOk  = initMpu();
  Imu  m      = {};
  bool readOk = mpuOk && readImu(m);
  float g     = sqrtf(m.ax * m.ax + m.ay * m.ay + m.az * m.az);
  bool whoOk  = whoAmI == 0x68 || whoAmI == 0x70 || whoAmI == 0x71 || whoAmI == 0x72 || whoAmI == 0x73;
  bool pass   = readOk && whoOk && g > 0.8f && g < 1.2f;

  char mpu[12] = "null", who[12] = "null";
  if (mpuAddr) {
    snprintf(mpu, sizeof(mpu), "\"0x%02x\"", mpuAddr);
    snprintf(who, sizeof(who), "\"0x%02x\"", whoAmI);
  }

  esp_chip_info_t ci;
  esp_chip_info(&ci);                            // revision is major*100+minor on core 3.x

  Serial.printf("{\"t\":\"selftest\",\"id\":\"%s\",\"fw\":\"" FW_NAME "\",\"chip\":\"%s\",\"rev\":%d,"
                "\"cores\":%d,\"flash\":%lu,\"psram\":%lu,\"sdk\":\"%s\",\"sda\":%d,\"scl\":%d,"
                "\"i2c\":[%s],\"mpu\":%s,\"whoami\":%s,\"g\":%.3f,\"pass\":%s}\n",
                deviceId().c_str(), ESP.getChipModel(), (int)ci.revision, (int)ci.cores,
                (unsigned long)ESP.getFlashChipSize(), (unsigned long)ESP.getPsramSize(), ESP.getSdkVersion(),
                SDA_PIN, SCL_PIN, found.c_str(), mpu, who, g, pass ? "true" : "false");
}

void handleCommand(const String &line) {
  if (line.indexOf("\"selftest\"") >= 0) {
    selftest();
  } else if (line.indexOf("\"hello\"") >= 0) {
    hello();
  } else if (line.indexOf("\"stream\"") >= 0) {
    streaming = line.indexOf("false") < 0;
    Serial.printf("{\"ok\":true,\"stream\":%s}\n", streaming ? "true" : "false");
  } else {
    Serial.println("{\"t\":\"err\",\"msg\":\"unknown command\"}");
  }
}

void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(400000);
  delay(50);
  hello();
  selftest();                                    // also configures the MPU
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      rxLine.trim();
      if (rxLine.length()) handleCommand(rxLine);
      rxLine = "";
    } else if (rxLine.length() < 256) {
      rxLine += c;
    }
  }

  uint32_t now = millis();
  if (!streaming || now - lastSample < 1000 / RATE_HZ) return;
  lastSample = now;

  Imu m;
  if (readImu(m)) {
    Serial.printf("{\"t\":\"imu\",\"s\":%lu,\"ax\":%.3f,\"ay\":%.3f,\"az\":%.3f,"
                  "\"gx\":%.2f,\"gy\":%.2f,\"gz\":%.2f,\"tc\":%.1f}\n",
                  (unsigned long)seq++, m.ax, m.ay, m.az, m.gx, m.gy, m.gz, m.tc);
  } else if (now - lastErr > 1000) {
    lastErr = now;
    Serial.println("{\"t\":\"err\",\"msg\":\"MPU-6050 read failed, retrying\"}");
    initMpu();
  }
}

      joystick_ball.ino · board "ESP32 Wrover Module" · no extra libraries
      // joystick_ball.ino
// Freenove ESP32-WROVER + HW-504 joystick demo for WROVER Flash Lab (Joystick 3D tab).
//
// Wiring (HW-504):
//   GND  -> GND
//   +5V  -> 3V3     NOT 5V. The stick outputs its own supply voltage, and ESP32 pins max out at 3.6 V.
//   VRx  -> GPIO 32 (ADC1)
//   VRy  -> GPIO 33 (ADC1)
//   SW   -> GPIO 13 (internal pull-up, reads LOW when pressed)
// ADC1 pins keep working with Wi-Fi on (ADC2 pins don't), and these three pins stay clear
// of the camera connector and the MPU-6050 on GPIO 21/22, so both demos can stay wired.
//
// Arduino IDE: Board "ESP32 Wrover Module". No extra libraries.
// Build:       Sketch -> Export Compiled Binary, then flash the *.merged.bin at 0x0.
//
// Serial protocol, 115200 baud, one JSON object per line:
//   board -> page  {"id":"wrover-a1b2c3","fw":"joystick-ball","ver":"1.0.0","caps":["joystick","button"]}
//                  {"t":"selftest", ...wiring report...}
//                  {"t":"joy","s":123,"x":1940,"y":1952,"sw":0}      x/y raw 0-4095, sw 1 = pressed
//   page -> board  {"cmd":"hello"}  {"cmd":"selftest"}  {"cmd":"stream","on":false}

#include <esp_chip_info.h>
#include <driver/gpio.h>

#define FW_NAME     "joystick-ball"
#define FW_VER      "1.0.0"
#define VRX_PIN     32
#define VRY_PIN     33
#define SW_PIN      13
#define RATE_HZ     50
#define DEBOUNCE_MS 15

bool     streaming = true;
uint32_t seq = 0, lastSample = 0, swChanged = 0;
bool     swState = false, swRaw = false;   // true = pressed
String   rxLine;

struct PinReport { int rest, noise, down, up; bool floating; };

String deviceId() {
  uint64_t mac = ESP.getEfuseMac();          // same factory MAC the web page reads over USB
  char id[16];
  snprintf(id, sizeof(id), "wrover-%02x%02x%02x",
           (uint8_t)(mac >> 24), (uint8_t)(mac >> 32), (uint8_t)(mac >> 40));
  return String(id);
}

int readAvg(int pin, int n) {
  long sum = 0;
  for (int i = 0; i < n; i++) sum += analogRead(pin);
  return sum / n;
}

// Wiring check: an unconnected ADC pin follows whichever pull resistor is switched on,
// while a pot wiper (a few kilohms) barely moves against the ~45 kOhm internal pulls.
PinReport testPin(int pin) {
  PinReport r;
  gpio_num_t g = (gpio_num_t)pin;
  analogRead(pin);                             // attach the pin to the ADC before touching pulls
  long sum = 0, sq = 0;
  for (int i = 0; i < 32; i++) { int v = analogRead(pin); sum += v; sq += (long)v * v; }
  r.rest  = sum / 32;
  r.noise = (int)sqrtf(fmaxf(0.0f, (float)sq / 32 - (float)r.rest * r.rest));
  gpio_pulldown_en(g); delay(5); r.down = readAvg(pin, 16); gpio_pulldown_dis(g);
  gpio_pullup_en(g);   delay(5); r.up   = readAvg(pin, 16); gpio_pullup_dis(g);
  delay(5);
  r.floating = (r.up - r.down) > 1500;
  return r;
}

void printPin(const char *key, const PinReport &r) {
  Serial.printf("\"%s\":{\"rest\":%d,\"noise\":%d,\"down\":%d,\"up\":%d,\"float\":%s}",
                key, r.rest, r.noise, r.down, r.up, r.floating ? "true" : "false");
}

void hello() {
  Serial.printf("{\"id\":\"%s\",\"fw\":\"" FW_NAME "\",\"ver\":\"" FW_VER "\",\"caps\":[\"joystick\",\"button\"]}\n",
                deviceId().c_str());
}

// Hardware report the page uses for its verification checklist. Keep hands off the stick while it runs.
void selftest() {
  PinReport x = testPin(VRX_PIN), y = testPin(VRY_PIN);
  int  swIdle = digitalRead(SW_PIN);
  bool midX = x.rest > 1400 && x.rest < 2700, midY = y.rest > 1400 && y.rest < 2700;
  bool pass = !x.floating && !y.floating && swIdle == HIGH && midX && midY;

  esp_chip_info_t ci;
  esp_chip_info(&ci);                          // revision is major*100+minor on core 3.x

  Serial.printf("{\"t\":\"selftest\",\"id\":\"%s\",\"fw\":\"" FW_NAME "\",\"chip\":\"%s\",\"rev\":%d,"
                "\"cores\":%d,\"flash\":%lu,\"psram\":%lu,\"sdk\":\"%s\",\"vrx\":%d,\"vry\":%d,\"swpin\":%d,",
                deviceId().c_str(), ESP.getChipModel(), (int)ci.revision, (int)ci.cores,
                (unsigned long)ESP.getFlashChipSize(), (unsigned long)ESP.getPsramSize(), ESP.getSdkVersion(),
                VRX_PIN, VRY_PIN, SW_PIN);
  printPin("x", x);
  Serial.print(",");
  printPin("y", y);
  Serial.printf(",\"sw_idle\":%d,\"cx\":%d,\"cy\":%d,\"pass\":%s}\n",
                swIdle, x.rest, y.rest, pass ? "true" : "false");
}

void handleCommand(const String &line) {
  if (line.indexOf("\"selftest\"") >= 0) {
    selftest();
  } else if (line.indexOf("\"hello\"") >= 0) {
    hello();
  } else if (line.indexOf("\"stream\"") >= 0) {
    streaming = line.indexOf("false") < 0;
    Serial.printf("{\"ok\":true,\"stream\":%s}\n", streaming ? "true" : "false");
  } else {
    Serial.println("{\"t\":\"err\",\"msg\":\"unknown command\"}");
  }
}

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);                    // 0-4095; default attenuation already covers 0-3.3 V
  pinMode(SW_PIN, INPUT_PULLUP);
  delay(50);
  hello();
  selftest();
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      rxLine.trim();
      if (rxLine.length()) handleCommand(rxLine);
      rxLine = "";
    } else if (rxLine.length() < 256) {
      rxLine += c;
    }
  }

  uint32_t now = millis();
  bool raw = digitalRead(SW_PIN) == LOW;
  if (raw != swRaw) { swRaw = raw; swChanged = now; }
  bool edge = false;
  if (swRaw != swState && now - swChanged >= DEBOUNCE_MS) { swState = swRaw; edge = true; }

  if (!streaming) return;
  if (!edge && now - lastSample < 1000 / RATE_HZ) return;   // button edges go out immediately
  lastSample = now;
  Serial.printf("{\"t\":\"joy\",\"s\":%lu,\"x\":%d,\"y\":%d,\"sw\":%d}\n",
                (unsigned long)seq++, readAvg(VRX_PIN, 4), readAvg(VRY_PIN, 4), swState ? 1 : 0);
}
